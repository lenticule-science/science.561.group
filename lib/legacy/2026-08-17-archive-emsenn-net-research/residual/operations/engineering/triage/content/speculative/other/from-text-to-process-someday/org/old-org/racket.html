1.3.1 plan give procedure

this procedure gives quantity of item to recipient from location - assuming all are valid

(though a new thing will be created for the recipient if necessary) 

(define (give #:date [date #f]
              #:item [item #f]
              #:quantity [quantity #f]
              #:location [location #f]
              #:recipient [recipient #f])
  (if location
      (if (hash-has-key? location item)
          (hash-set! location item (- (hash-ref location item) quantity))
          (log-error "Tried to give something from somewhere it doesn't exist"))
      (log-error "No location specified")))

1.3.2 plan receive procedure

(define (receive #:date [date #f]
                 #:item [item #f]
                 #:quantity [quantity #f]
                 #:location [location #f]
                 #:giver [giver #f])
  (if location
      (if (hash-has-key? location item)
          (hash-set! location item (+ (hash-ref location item) quantity))
          (hash-set! location item quantity))
      (display "No location specified.\n")))

1.3.3 plan receive-multiple-items procedure

(define (receive-multiple-items #:date [date #f]
                                #:location [location #f]
                                #:giver [giver #f]
                                #:items [items #f])
  (if items
      (map (lambda (item)
             (receive #:date date
                      #:location location
                      #:giver giver
                      #:item (car item)
                      #:quantity (cdr item)))
           items)
      (display "No items specified.\n")))
