\section{Lifting settling to valuations}
\begin{proposition}[Pointwise valuation settling]\label{thm:valuation-settle}\lean{Obsmetrologia.Valuation.settle}\leanok\uses{thm:settling, thm:valuation}
\texttt{Valuation.settle op v} applies a charge-closure pointwise to a whole
valuation, $\varphi \mapsto \texttt{op}(v\,\varphi)$, lifting the single-charge
fixed-point dynamic to entire traces.
\end{proposition}
\begin{theorem}[Settling the knowledge-least valuation]\label{thm:settle-knowledgeleast}\lean{Obsmetrologia.Valuation.settle_knowledgeLeast}\leanok\uses{thm:valuation-settle, thm:knowledge-least, thm:valuation, thm:join-closure}
Settling the knowledge-least valuation toward a target charge $k$ yields the
constant-$k$ valuation: the fixed point the modal dynamic climbs to from the
seed the positing deposited.
\end{theorem}
