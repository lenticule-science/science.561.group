\subsection{Possibilistic support models}

\begin{proposition}[Measurement scenario]\label{thm:measurement-scenario}\lean{Obsmetrologia.QuantumContextuality.Scenario}\leanok\uses{thm:bitopological-site-grothendieck, thm:sheaf-cover, thm:sheaf-separated}
A measurement scenario is a finite cover of co-measurable contexts.
\end{proposition}

\begin{proposition}[Possibilistic empirical model]\label{thm:possibilistic-model}\lean{Obsmetrologia.QuantumContextuality.Model}\leanok\uses{thm:measurement-scenario}
A model assigns a support predicate to each context. Contextuality is failure
of those local supports to glue to a global section.
\end{proposition}

\begin{theorem}[The PR box is strongly contextual]\label{thm:pr-box-strong-contextual}\lean{Obsmetrologia.QuantumContextuality.prBox_strongly_contextual}\leanok\uses{thm:possibilistic-model}
The PR support model has every context inhabited but has no global section.
\end{theorem}
