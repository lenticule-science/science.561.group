\section{Measurement retains articulated mark provenance}

\begin{theorem}[Measured poles are witnessed by articulated marks in the trace]\label{thm:measured-mark-provenance}\notready\uses{thm:trace-measure, thm:no-denial, thm:predication-mark-provenance}
\emph{Proof-line frontier.} A positive measured pole has an articulating in the trace whose oriented cell is the queried cell; a negative pole has one whose oriented cell is its contradictory. In both cases the relevant primitive marks occur in the trace-level articulated mark record.
\end{theorem}
