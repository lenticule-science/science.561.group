\subsection{Liquid closures}

\begin{theorem}[DFT minimizers are equilibria]\label{thm:dft-minimizer-equilibrium}\lean{Obsmetrologia.KineticTheory.Liquid.minimizer_is_equilibrium}\leanok\uses{thm:euler-e-internal-real}
In the finite density-functional projection, a free-energy minimizer is an
equilibrium of the relaxation flow.
\end{theorem}

\begin{theorem}[Ornstein--Zernike total correlation is harmonic]\label{thm:oz-total-correlation-harmonic}\lean{Obsmetrologia.KineticTheory.Liquid.totalCorrelation_harmonic}\leanok\uses{thm:tarski-sheaf-closure}
The finite Ornstein--Zernike closure presents total correlation as the
harmonic fixed point of the closure complex.
\end{theorem}
