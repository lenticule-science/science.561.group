\subsubsection{Quadratic-surd engine}

\begin{theorem}[Every quadratic surd is a real root of its minimal quadratic]\label{thm:gap-quad-surd}\lean{Obsmetrologia.Real.Internal.quadratic_surd_sub_root}\leanok\uses{thm:tile4-irrational-gap}
For $1\le d\le a\cdot a$, the value $a-\sqrt d:\mathbb R$ is a root of
$x^2-2ax+(a^2-d)$, equivalently
$g^2+(a^2-d)=(a+a)g$, division-free over the constructive reals. This is the
generic algebraic carrier behind the crystallographic family
(\cref{thm:tile-crystallographic-gap}, $a=2$) and the golden 5-site gap
(\cref{thm:tile5-golden-gap}, $a=3$).
\end{theorem}
