\subsubsection{General spectral-gap frontier}

\begin{theorem}[The general tile spectral gap value $\lambda_1$]\label{thm:tile-spectral-gap}\notready\uses{thm:tile-rayleigh-bottom, thm:two-site-gap, thm:three-site-gap-lower, thm:tile-harmonic-kernel-constants, thm:tile4-irrational-gap, thm:tile-crystallographic-gap, thm:gap-quad-surd, thm:tile5-golden-gap, thm:tile7-cubic-gap, thm:tile7-gap-bracket, thm:euler-e-internal-real}
\emph{Open frontier: the general-$n$ value needs the real infimum and cosine
eigenbasis.} Positivity of the gap holds for all $n$, small rational gaps are
known for $N=2,3$, every quadratic-gap tile through $N=6$ is realized as an
explicit integer quadratic root, and the first cubic escape $N=7$ is pinned by
an explicit cubic and an IVT bracket. Two threads remain: constructing the
$N=7$ gap itself as a Bishop real by rational bisection of its bracket, and
proving the general closed form
$\lambda_1=2(1-\cos(\pi/N))$ from the cosine eigenbasis of the path Laplacian.
\end{theorem}
