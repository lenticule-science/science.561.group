\section{Predication is generated from articulated marks}

\begin{theorem}[Predication descends from articulated marks without recovering history]\label{thm:predication-mark-provenance}\notready\uses{thm:articulating-mark-provenance, thm:predication, thm:oriented-cell}
\emph{Proof-line frontier.} Every coordinate mark of a generated oriented cell is supported by its articulating. Every predication cell has a synthetic articulated source made only from its own marks. The oriented-cell map is not injective, so the synthetic section proves generability but not historical provenance recovery.
\end{theorem}
