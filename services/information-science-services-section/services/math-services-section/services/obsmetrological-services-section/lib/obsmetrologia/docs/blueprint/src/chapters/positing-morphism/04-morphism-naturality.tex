\section{Generator-graded naturality: involution, injective, projection}

\cref{thm:posit-converse-natural} proved that \texttt{posit} is natural under
reindexing along \texttt{converse}. This section works out, and proves, the
honest naturality statement for each of the other predication morphism
generators of \cref{thm:predication-proj,thm:predication-specialize}. The
four generators do not all deserve the same strength of claim, and that is
the point of this section:

\begin{itemize}
\item \textbf{converse} --- an involution (hence a bijection). Full pointwise
naturality, already established.
\item \textbf{specialize \(f\), \(f\) injective} --- an injective but not in
general surjective or involutive reindexing. Full pointwise naturality still
holds, because injectivity alone is the load-bearing hypothesis: the proof
only ever uses the cancellation direction of \(f\), never an inverse.
\item \textbf{subjProj / objProj} --- idempotent, non-injective projections
onto the diagonal. The single-posit naturality form \textbf{fails in
general} --- a concrete counterexample is exhibited, not merely asserted.
What holds instead is the narrower \emph{diagonal fixed-point} case.
\end{itemize}

\begin{theorem}[The exchange lemma --- true for every reindexing map]\label{thm:reindex-posit-exchange}\lean{Obsmetrologia.Valuation.reindex_posit_exchange}\leanok\uses{thm:reindex, thm:posit, thm:delta}
For \emph{any} \(r : \mathtt{Claim} \to \mathtt{Claim}\) --- no injectivity,
no involution, nothing assumed --- reindexing a posited valuation along \(r\)
is the join of the reindexed valuation with \texttt{delta} evaluated at the
pushed-forward point \(r\,\psi\): \(r^{*}(\texttt{posit}\,\varphi\,q\,v)\,\psi
= (r^{*}v)\,\psi \vee_k \delta_{\varphi,q}(r\,\psi)\). Immediate from the
definitions; every naturality statement below specializes this one fact.
\end{theorem}

\subsection*{Injective reindexing: full naturality}

\begin{proposition}[Pull-back form of \texttt{delta}]\label{thm:delta-reindex-point}\lean{Obsmetrologia.Valuation.delta_reindex_point}\leanok\uses{thm:delta}
Unconditional in \(r\): a restatement of \texttt{delta}'s definition at the
pushed-forward point \(r\,\psi\).
\end{proposition}

\begin{theorem}[Naturality under injective reindexing]\label{thm:posit-reindex-natural-injective}\lean{Obsmetrologia.Valuation.posit_reindex_natural_of_injective}\leanok\uses{thm:reindex-posit-exchange, thm:delta-reindex-point}
When \(r\) is injective and the posited cell \(\varphi\) is itself \(r\)'s
image of some \(\varphi'\), reindexing the posit along \(r\) is the same as
positing at \(\varphi'\) into the already-reindexed valuation. The only use
of injectivity is the cancellation direction \(r\,\psi = r\,\varphi' \to \psi
= \varphi'\); no inverse of \(r\) is constructed or needed.
\end{theorem}

\begin{proposition}[\texttt{specialize f} is injective on cells iff \(f\) is injective]\label{thm:specialize-injective-iff}\lean{Obsmetrologia.PredicationCell.specialize_injective_iff}\leanok\uses{thm:predication-specialize}
The entity coordinates of a predication cell pass through
\texttt{specialize} unchanged, so all discriminating power sits in the
predicate coordinate: \texttt{specialize}~\(f\) is injective on cells iff
\(f\) is injective on predicates.
\end{proposition}

\begin{theorem}[Naturality under an injective specialization]\label{thm:posit-reindex-specialize-injective}\lean{Obsmetrologia.Valuation.posit_reindex_natural_of_specialize_injective}\leanok\uses{thm:posit-reindex-natural-injective, thm:specialize-injective-iff}
For \(f\) injective on predicates, reindexing a \texttt{specialize f}-posit
is a single posit at the pulled-back cell \(\varphi'\) --- a direct corollary
of \cref{thm:posit-reindex-natural-injective} via
\cref{thm:specialize-injective-iff}.
\end{theorem}

\subsection*{Non-injective projections: the single-posit form fails; the diagonal form holds}

\texttt{subjProj} and \texttt{objProj} are idempotent projections onto the
diagonal \(\{\chi : \chi.\mathrm{subject} = \chi.\mathrm{object}\}\), and
neither is injective: every cell sharing a subject (resp.\ object) and
predicate collapses to the same image regardless of the other coordinate.
So for a target \(\varphi\) in the image, the fiber of \texttt{subjProj}
over \(\varphi\) is not a singleton, and no single \(\varphi'\) can make the
single-posit naturality equation hold at every point of that fiber.

\begin{theorem}[The single-posit form fails for \texttt{subjProj}, in general]\label{thm:posit-reindex-subjproj-not-natural}\lean{Obsmetrologia.Valuation.posit_reindex_subjProj_not_natural}\leanok\uses{thm:reindex-posit-exchange, thm:predication-proj, thm:knowledge-least}
No \(\varphi'\) makes
\(\texttt{reindex}\,\texttt{subjProj}\,(\texttt{posit}\,\varphi_0\,q\,v)\,\psi
= \texttt{posit}\,\varphi'\,q\,(\texttt{reindex}\,\texttt{subjProj}\,v)\,\psi\)
hold at every \(\psi\), witnessed concretely: two cells \(\psi_1 \ne \psi_2\)
sharing subject and predicate but differing in object both lie in
\texttt{subjProj}'s fiber over the same \(\varphi_0\), so matching the
equation at \(\psi_1\) forces \(\varphi' = \psi_1\) and at \(\psi_2\) forces
\(\varphi' = \psi_2\) --- impossible since \(\psi_1 \ne \psi_2\). This is the
honest negative counterpart to \cref{thm:posit-reindex-natural-injective}:
it is exactly the failure of injectivity that breaks the single-posit form.
No positive general-form theorem for \texttt{subjProj} (or, symmetrically,
\texttt{objProj}) is claimed.
\end{theorem}

\begin{proposition}[The projections fix the diagonal]\label{thm:proj-fixed-diagonal}\lean{Obsmetrologia.Valuation.subjProj_fixed_of_diagonal, Obsmetrologia.Valuation.objProj_fixed_of_diagonal}\leanok\uses{thm:predication-proj}
A cell whose subject already equals its object is untouched by
\texttt{subjProj} (resp.\ \texttt{objProj}).
\end{proposition}

\begin{theorem}[The honest positive form, restricted to the diagonal]\label{thm:posit-reindex-proj-diagonal}\lean{Obsmetrologia.Valuation.posit_reindex_subjProj_diagonal, Obsmetrologia.Valuation.posit_reindex_objProj_diagonal}\leanok\uses{thm:proj-fixed-diagonal, thm:reindex-posit-exchange}
Restricted to query points \(\psi\) already on the diagonal, \texttt{subjProj}
(resp.\ \texttt{objProj}) acts as the identity there
(\cref{thm:proj-fixed-diagonal}), so the single-posit naturality form holds
--- for \emph{every} \(\varphi\), not only diagonal \(\varphi\): if
\(\varphi\) is off the diagonal, \(\texttt{subjProj}\,\psi\) (\(=\psi\),
diagonal) can never equal it, so both sides independently collapse to the
reindexed valuation at \(\psi\) with no contribution from the posit, and the
equation still holds trivially. This replaces the failed general claim of
\cref{thm:posit-reindex-subjproj-not-natural} with the true, narrower fact.
\end{theorem}
