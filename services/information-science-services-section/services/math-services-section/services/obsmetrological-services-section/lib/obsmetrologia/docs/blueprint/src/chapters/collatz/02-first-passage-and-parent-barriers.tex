\section{First passage and parent barriers}

\begin{theorem}[A first-passage word ends in E]
  \label{thm:collatz-last-E}
  \lean{Obsmetrologia.Closure.Collatz.firstPassage_last_E}
  \leanok
  \uses{lem:collatz-trace-powers}
  A \emph{first-passage word} is the first prefix at which the trace turns contracting
  ($A<B$), every proper prefix noncontracting. Any nonempty first-passage word ends in the
  crossing letter $E$: an $O$-ending would force $3A<2B\le 2A$ against $A>0$.
\end{theorem}
\begin{theorem}[First passage iff parent barrier]
  \label{thm:collatz-firstpassage-iff}
  \lean{Obsmetrologia.Closure.Collatz.firstPassage_append_E_iff_parentBarrier}
  \leanok
  \uses{thm:collatz-last-E}
  $p \mathbin{+\!\!+} [E]$ is first-passage iff $p$ is a \emph{parent barrier}:
  $B_p\le A_p<2B_p$ with every proper prefix noncontracting. This is the structural pivot of
  the analysis.
\end{theorem}
\begin{lemma}[Exponent sandwich at the crossing]
  \label{lem:collatz-power-bounds}
  \lean{Obsmetrologia.Closure.Collatz.firstPassage_power_bounds}
  \leanok
  \uses{thm:collatz-last-E}
  At a first-passage word $w$: $2^{|w|-1}\le 3^{\,\mathrm{oddCount}\,w}<2^{|w|}$.
\end{lemma}
