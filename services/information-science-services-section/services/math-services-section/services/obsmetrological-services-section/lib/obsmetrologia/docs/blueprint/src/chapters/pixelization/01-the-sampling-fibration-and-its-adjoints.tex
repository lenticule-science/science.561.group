\section{The sampling fibration and its adjoints}

This section records claim nodes and dependency edges.

\begin{proposition}[Sampling fibration]\label{thm:sampling-fibration}\lean{Obsmetrologia.Valuation.SamplingFibration}\notready\uses{thm:finiteFibration}
\emph{(Open frontier --- stated, not formalized.)} The two-carrier generalization of
\cref{thm:finiteFibration}: a base map $\texttt{base} : \texttt{Domain} \to
\texttt{GridPoint}$ from a fine carrier to a coarse lattice, with each grid point's
preimage presented as a finite list and the membership law $x \in
\texttt{fiber}\,p \iff \texttt{base}\,x = p$. The existing proofs of
\cref{thm:finiteFibration} use nothing specific to a single carrier; the obligation is
the mechanical re-typing, with $\texttt{Fin}\,n \times \texttt{Fin}\,m$ (pixels) and
hexagonal coordinates as intended instances.
\end{proposition}

\begin{theorem}[Pixelize is the left adjoint; the box filter is the pullback leg]\label{thm:pixelize-adjoint-triple}\lean{Obsmetrologia.Valuation.pixelize, Obsmetrologia.Valuation.boxReconstruct, Obsmetrologia.Valuation.pixelize_adjunction, Obsmetrologia.Valuation.boxReconstruct_adjunction}\notready\uses{thm:sampling-fibration, thm:existsF-adjunction, thm:forallF-adjunction}
\emph{(Open frontier --- stated, not formalized.)} The adjoint triple $\exists_{\texttt{base}}
\dashv \texttt{base}^{*} \dashv \forall_{\texttt{base}}$ of
\cref{thm:existsF-adjunction}/\cref{thm:forallF-adjunction}, restated over a sampling
fibration: $\texttt{pixelize} := \exists_{\texttt{base}}$ (fiber knowledge-join, the
honest lossy aggregation) and $\texttt{boxReconstruct} := \texttt{base}^{*} =
\texttt{reindex}$, which is \emph{forced} to be constant on each fiber --- pullback along
a many-to-one map repeats one value across the fiber. The ``little square'' is not a
filter choice but the unique map the pullback direction admits; ideal (sinc)
reconstruction is neither fiber-local nor a pullback (non-monotone in the pointwise
order) and is therefore outside the \emph{position-side} calculus by construction ---
it reappears as the fundamental-domain transversal of the dual fibration,
\cref{thm:spectral-nyquist}. The proofs are the existing adjunction proofs at the new
types.
\end{theorem}

\begin{proposition}[Constant-fiber collapse]\label{thm:sample-reconstruct-id}\lean{Obsmetrologia.Valuation.sample_reconstruct_id}\notready\uses{thm:sampling-fibration, thm:charge-joinList}
\emph{(Open frontier --- stated, not formalized.)} For a sampling fibration with nonempty
fibers, $\texttt{pixelize} \circ \texttt{boxReconstruct} = \texttt{id}$: the fiber
knowledge-join of a constant list is that constant. One induction on the fold of
\cref{thm:charge-joinList} with idempotent $\vee_k$; the sole genuinely new lemma the
closure below requires.
\end{proposition}
