\subsection{Aperiodic settling frame}

\begin{theorem}[The aperiodic witness as a settling frame --- the $\kappa \geq \omega_1$ carrier]\label{thm:aperiodic-settle-frame}\lean{Obsmetrologia.AperiodicOrder.Tiling.AperiodicCarrier.aperiodic_settle_frame}\leanok\uses{thm:chromatic-lorentzian-carrier, thm:coalgebraic-crossbase-mld}
The aperiodic witness becomes a settling frame by equipping it with inflation
stage, non-idempotent step, and \(\mathrm{cl\text{-}inv}\) as direct
\(\mathrm{settle}\).  Inflation preserves \(\mathrm{cl\text{-}inv}\), so
closure-absorption holds by construction.  The frame avoids an ordinal iteration
wall and carries geometric MLD \(\iff\) cohomological equivalence in the
aperiodic regime.
\end{theorem}

\begin{proposition}[History system]\label{thm:aperiodic-history-system}\lean{Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.HistorySystem}\leanok\uses{thm:aperiodic-settle-frame}
A history system is the aperiodic carrier's ordered action spine: an empty
history, concatenation, and a prefix-extension order \(t \leq t'\) exactly when
\(t'\) is obtained from \(t\) by concatenating a further history.
\end{proposition}

\begin{theorem}[Prefix laws for histories]\label{thm:aperiodic-history-prefix-laws}\lean{Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.HistorySystem.empty_le, Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.HistorySystem.le_concat, Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.HistorySystem.le_trans, Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.HistorySystem.concat_left_monotone}\leanok\uses{thm:aperiodic-history-system}
The empty history lies below every history, every concatenation \(t\cdot u\)
lies above \(t\), prefix extension is transitive, and left-concatenation is
monotone for the prefix order.
\end{theorem}

\begin{theorem}[Cancellation and unit-freeness make prefix antisymmetric]\label{thm:prefix-antisymm-cancel-unitfree}\lean{Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.prefix_antisymm_of_cancel_unit_free}\leanok\uses{thm:aperiodic-history-prefix-laws}
For any concatenating carrier, left-cancellation plus unit-freeness force the
prefix-extension relation to be antisymmetric. This is the dependency-free
fragment of the archived relational-hyperverse prefix-order argument.
\end{theorem}

\begin{proposition}[Commutation relation]\label{thm:history-commutation-relation}\lean{Obsmetrologia.AperiodicOrder.Tiling.Aperiodic.CommutationRelation}\leanok\uses{thm:aperiodic-history-system}
The history quality facet is a symmetric irreflexive relation on generating
steps, saying which adjacent steps commute without changing the resulting
history.
\end{proposition}

\begin{theorem}[The quotient history monoid is universal]\label{thm:history-quotient-universality}\notready\uses{thm:history-commutation-relation, thm:prefix-antisymm-cancel-unitfree}
The full quotient \(M(\Sigma,I)\) by adjacent commuting swaps should be the
initial history system. The archive proves the swap congruence, unit-freeness,
and the independent-step diamond, but explicitly leaves left-cancellation open;
the active proof needs Foata-normal-form uniqueness or an equivalent
cancellation theorem to close this node.
\end{theorem}
