\subsection{Continuum colimits}

\begin{theorem}[Abstract directed-join interchange for monotone $\mathbb Z$-functionals]\label{thm:monotone-functional-limit-attained}\lean{Obsmetrologia.CausalSets.Continuum.Colimit.monotone_functional_limit_attained}\leanok\uses{thm:omega-limit-is-finite-settled}
Over the finite charge lattice, any $\le_k$-monotone
$F:\texttt{Charge}\to\mathbb Z$ on a \texttt{DirectedFamily} attains its value at
the directed join at a member of the family.  This is an abstract finite-carrier
lemma; the action-specific continuum claim is \cref{thm:bd-action-continuum-colimit}.
\end{theorem}

\begin{theorem}[The Benincasa--Dowker action's continuum limit is a colimit]\label{thm:bd-action-continuum-colimit}\notready\uses{thm:monotone-functional-limit-attained, thm:sym-trace-is-ricci-scalar}
\emph{(Open frontier --- stated, not formalized.)} The intended continuum claim
is that the $V\to\infty$ Benincasa--Dowker action over a growing sprinkling is the
directed-join colimit of \cref{thm:monotone-functional-limit-attained}.  The
missing fibers are an unbounded sprinkling/refinement tower and the actual action
functional on that tower.
\end{theorem}

\begin{theorem}[The Chu geometry's information action is an attained colimit]\label{thm:chu-info-action-colimit}\lean{Obsmetrologia.CausalSets.Continuum.ChuActionColimit.chu_info_action_continuum_attained}\leanok\uses{thm:monotone-functional-limit-attained}
The colimit mechanism is proved for the finite FOUR-valued information action of a
Chu geometry.  The observed-knowledge tower is a concrete \texttt{RefinementTower}
and the information action is attained at a finite stage.  The M\"obius
Benincasa--Dowker action and metric Ricci magnitude remain the red fibers of
\cref{thm:bd-action-continuum-colimit} and \cref{thm:sym-trace-is-ricci-scalar}.
\end{theorem}

\begin{theorem}[The constructive continuum certificate is an attained colimit below $\omega$]\label{thm:constructive-continuum-colimit-certificate}\lean{Obsmetrologia.CausalSets.Continuum.Colimit.constructive_continuum_colimit_attained_below_omega, Obsmetrologia.CausalSets.Continuum.ChuActionColimit.chu_constructive_continuum_colimit_attained_below_omega}\leanok\uses{thm:monotone-functional-limit-attained, thm:reals-close-value-settling-hub, thm:real-stochastic-interval-value-category, thm:chu-info-action-colimit}
The continuum carrier the formalization proves is constructive:
finite FOUR-valued refinement towers attain their directed join at the finite
ordinal stage \texttt{ofNat N} below \(\omega\), the Bishop reals are the
\(\kappa=\omega\) Scott-continuous completion of the rationals, and the real
stochastic interval \([0,1]\) is a value category over those reals.  This is the
precise replacement for any set-cardinality reading: no cardinal-comparison
surface is introduced here, so the Continuum Hypothesis is not a theorem, axiom,
or obligation of this continuum mechanism.
\end{theorem}
