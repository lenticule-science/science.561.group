\subsection{Residual curvature}

\begin{theorem}[The Lambek bi-residuation bundle]\label{thm:lambek-biresiduation}\lean{Obsmetrologia.Lambek.lambek_biresiduation}\leanok\uses{thm:converse-adjunction, thm:four-bilattice}
From a \texttt{LambekAlgebra}'s two adjunction fields, the full
substructural/algebraic correspondence follows: both adjunctions, left/right
modus ponens, and two-sided fusion monotonicity.  The node is the finite
residuated carrier used by the curvature defects below.
\end{theorem}

\begin{theorem}[Flatness is commutativity of the two residuals]\label{thm:residual-coincidence-iff-commutative}\lean{Obsmetrologia.CausalSets.Curvature.Lambek.residuals_coincide_iff_commutative}\leanok\uses{thm:lambek-biresiduation}
In a residuated lattice, $a\backslash c=c/a$ for all $a,c$ iff the product
commutes.  Thus the residual defect is the non-commutativity fiber.  The
spacetime-curvature reading is carried only by the red geometric nodes.
\end{theorem}

\begin{theorem}[Antisymmetric M\"obius trace conservation (discrete Bianchi)]\label{thm:mobius-tensor-total-zero}\lean{Obsmetrologia.CausalSets.Curvature.Ricci.tensor_total_zero}\leanok\uses{thm:causal-cochain-complex}
For $T(x,y)=\mu(y,x)-\mu(x,y)$, the total $\sum_{x,y}T(x,y)$ is zero.  This is
the antisymmetry identity for the finite incidence carrier, proved by exchanging
summation order.
\end{theorem}

\begin{theorem}[Symmetric M\"obius trace totals to twice the action]\label{thm:sym-mobius-trace-two-action}\lean{Obsmetrologia.CausalSets.Curvature.Ricci.total_ricciScalar_eq_two_bdAction}\leanok\uses{thm:mobius-tensor-total-zero}
For $\mathrm{sym}(x,y)=\mu(y,x)+\mu(x,y)$, the total row trace is
$2\cdot\mathrm{bdAction}$.  This is an exact incidence identity.  The
geometric Ricci-scalar interpretation is the separate red node
\cref{thm:sym-trace-is-ricci-scalar}.
\end{theorem}

\begin{theorem}[The symmetric M\"obius trace is the continuum Ricci scalar]\label{thm:sym-trace-is-ricci-scalar}\notready\uses{thm:sym-mobius-trace-two-action, thm:myrheim-meyer-dimension}
\emph{(Open frontier --- conjecture, not formalized.)} Under a faithful Poisson
sprinkling into a Lorentzian manifold, the metric-weighted continuum scaling of
the symmetric M\"obius trace should recover the Riemannian Ricci scalar $R$.
\Cref{thm:sym-mobius-trace-two-action} supplies only the discrete identity; the
missing fibers are the manifold, metric density, sprinkling average, and scaling
law.
\end{theorem}
