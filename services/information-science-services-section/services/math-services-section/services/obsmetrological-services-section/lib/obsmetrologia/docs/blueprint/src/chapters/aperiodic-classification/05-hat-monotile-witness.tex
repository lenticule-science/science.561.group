\section{Hat monotile witness}

\begin{theorem}[The hat metatile substitution is a primitive integer witness with golden Perron factor]\label{thm:hat-metatile-substitution}\lean{Obsmetrologia.AperiodicOrder.Tiling.Witness.hat_metatile_witness}\leanok\uses{thm:stepn-chu-cayley-hamilton, thm:metallic-family-cayley-hamilton}
The hat metatile substitution $H'=3H+T+3P+3F$, $T'=H$, $P'=2H+P+2F$, $F'=2H+P+3F$ (Smith--Myers--Kaplan--Goodman-Strauss; the face substitution of Baake--G\"ahler--Sadun at $r=1$) abelianizes to
$M=\begin{pmatrix}3&1&3&3\\1&0&0&0\\2&0&1&2\\2&0&1&3\end{pmatrix}$.
$M$ is nonnegative and primitive ($M^2>0$), satisfies $x^4-7x^3+7x-1=(x^2-7x+1)(x^2-1)$, and its Perron factor $x^2-7x+1$ annihilates $F^4$ for the Fibonacci matrix $F$: the hat's area inflation $\varphi^4$ is the fourth rung of the metallic tower. $M$ is the $\mathrm{pos}$-incidence of a FOUR-valued Chu relation on the metatile alphabet.
\end{theorem}

\begin{theorem}[A finite cohomology witness for the hat hull ($\check H^1=\mathbb{Z}^4$)]\label{thm:hat-hull-cohomology}\notready\uses{thm:hat-metatile-substitution, thm:jeandel-rao-witness}
\emph{Open frontier.} The \v{C}ech cohomology of the hat hull is $\check H^0=\mathbb{Z}$, $\check H^1=\mathbb{Z}^4$, $\check H^2=\mathbb{Z}^{10}$ (Baake--G\"ahler--Sadun, Theorem 4). The target is a finite Anderson--Putnam complex of the hat transcribed on the Jeandel--Rao pattern, with a nonzero $H^1$ class exhibited as a cocycle and its persistence through the substitution as a chain map.
\end{theorem}
