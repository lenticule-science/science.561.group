\section{The four-corner coherence cell at successive depths}

This section records claim nodes and dependency edges.

\begin{theorem}[The modal four]\label{thm:modal}\lean{Obsmetrologia.Modal}\leanok\uses{thm:four-object, thm:quartet-corner}
\texttt{Modal} is the four-constructor inductive $\{\texttt{neither}, \texttt{dia}, \texttt{box}, \texttt{both}\}$ --- the $\{$neither, $\Diamond$, $\Box$, both$\}$ reading of the recurring FOUR, naming the four ways of acting on the free carrier with two commuting closures. Its bijection \texttt{Modal.cornerBij} with \texttt{Articulating.Corner} certifies that the modal four-corner is the \emph{same} object as the articulation phase, the bilattice charge, and the nuclear-quartet corner, re-met under modal names.
\end{theorem}
\begin{theorem}[The master conflation --- the keystone]\label{thm:quartet-recurrence}\lean{Obsmetrologia.quartet_recurrence}\leanok\uses{thm:modal, thm:articulation-phase-is-charge, thm:four-fold}
\texttt{QuartetRecurrence} bundles the round-trip witnesses, and \texttt{quartet\_recurrence} proves it: the articulation-cell phase, \texttt{Articulating.FourCorner} ($\texttt{Bool} \times \texttt{Bool}$), \texttt{Charge} ($=\texttt{Articulating.Corner}$), the nuclear-quartet corner index, and \texttt{Modal} are \emph{all one object}. Every reading's round-trip is the identity, the articulation phase is the charge, and the cell is exhaustively four-cornered with no fifth reading. This is the one four-corner cell met at successive depths, made a single theorem depending on no axioms beyond \texttt{propext} and \texttt{Quot.sound}.
\end{theorem}
\begin{proposition}[The abstract commuting pair]\label{thm:commuting-pair}\lean{Obsmetrologia.CommutingPair}\leanok\uses{thm:quartet-recurrence}
A \texttt{CommutingPair} on a carrier $C$ is two endomaps $f, g$ with $f \circ g = g \circ f$ --- the bare data shared by twistoring (two involutions) and commutating (two nuclei). The involution and idempotency laws of the specific depth are deliberately absent: they are not part of the shared coherence cell. Its \texttt{CommutingPair.cornerOp} reads the four corners as identity, $f$ alone, $g$ alone, and $f \circ g$.
\end{proposition}
\begin{theorem}[The recurring coherence 2-cell]\label{thm:commuting-square}\lean{Obsmetrologia.CommutingPair.square}\leanok\uses{thm:commuting-pair}
\texttt{CommutingPair.square} proves the generic coherence: the \texttt{stable} corner operator $f \circ g$ agrees with $g \circ f$ --- both paths from the free corner to the doubly-acted corner coincide. This single lemma is the commuting-square 2-cell that recurs at every depth of the spine.
\end{theorem}
\begin{theorem}[Twistoring equals commutating]\label{thm:twistoring-eq-commutating}\lean{Obsmetrologia.twistoring_eq_commutating}\leanok\uses{thm:commuting-square, thm:klein-four-symmetry, thm:nuclear-quartet}
\texttt{twistoringPair} instantiates the cell on \texttt{Charge} with $f = {\sim}$, $g = {-}$, commuting by the $K_4$ law; \texttt{commutatingPair} instantiates it on a Heyting algebra with $f = j_1$, $g = j_2$, commuting by the nuclear square. \texttt{twistoring\_eq\_commutating} proves both satisfy the \emph{same} generic \texttt{CommutingPair.square} at every point and index the same exhaustive four corners. The $\sigma\circ\Delta = \Delta\circ\sigma$ commuting square at aperiodic settling and the ${\sim}\circ{-} = {-}\circ{\sim}$ twist at the articulating birth-corner are one coherence 2-cell. The depth-specific laws differ --- involutions at articulating, idempotent nuclei at settling --- but neither enters the shared cell, which is only the commuting square.
\end{theorem}

\noindent\emph{Conflation frontier.} Three places where the formalization currently states a structured equivalence as a bare bijection / a single worked instance --- the data is named for the object it builds (\texttt{regimeCornerBij}, \texttt{fcaClosureOp}, \texttt{frobeniusTqftCorrespondence}), and the \emph{specific} relation each only gestures at is the open node below.

\begin{theorem}[The fundamental cycle, transported onto the regimes]\label{thm:regime-spin}\lean{Obsmetrologia.AperiodicOrder.Tiling.regimeSpin, Obsmetrologia.AperiodicOrder.Tiling.regimeSpin_cycle, Obsmetrologia.AperiodicOrder.Tiling.regimeSpin_four, Obsmetrologia.AperiodicOrder.Tiling.regimeSpin_regimeSpin}\leanok\uses{thm:quartet-recurrence}
\texttt{AperiodicOrder.Tiling.spin} (\texttt{Percolation.lean}) --- the order-4 rotation \texttt{D8.negSwap}, percolation's load-bearing fundamental cycle --- and \texttt{regimeCornerBij} were built independently, in unrelated modules, for unrelated reasons. Transported through the bijection, \texttt{regimeSpin} $:= \texttt{ofCorner} \circ \texttt{spin} \circ \texttt{toCorner}$ is a genuine order-4 rotation of \texttt{ClosureRegime} itself, and its cycle is \emph{not} the $\kappa$-height chain: one spin sends strict$\mapsto$scott$\mapsto$aperiodic$\mapsto$transfinite-bounded$\mapsto$strict, skipping straight from scott to aperiodic. Squaring it gives a clean involution --- strict$\leftrightarrow$aperiodic (the bilattice's two extremes, $\bot_k$ and the fully-alive $\top_k$) and scott$\leftrightarrow$transfinite-bounded (the two genuinely-infinite, non-extremal regimes) --- generated by the same order-exchanging symmetry (\texttt{spin\_exchanges\_orders}) that produces the fundamental cycle in the first place, not chosen after the fact to make the pairing come out clean. This is one generator of the Klein-four/D8 gauge carried onto the regime side and shown to act non-trivially; it is not yet the full gauge action the node below asks for.
\end{theorem}
\begin{theorem}[The regime is the fifth quartet member]\label{thm:regime-quartet-member}\notready\uses{thm:quartet-recurrence, thm:regime-spin}
\emph{(Open frontier --- stated, not formalized.)} The closure-ordinal regime is conflated with the FOUR by an explicit bijection \texttt{regimeCornerBij} (strict$\mapsto$absurd, aperiodic$\mapsto$stable), and \texttt{regime\_is\_four\_corner} proves the round-trips --- but the regime is \emph{not} yet a member of the master conflation (\cref{thm:quartet-recurrence}), which bundles only Articulating.Cell/Articulating.FourCorner/Charge/NuclearQuartet/Modal. Folding it in needs the \emph{specific} structure the bijection transports, and that is subtle: the $\kappa$-stratification is a \textbf{chain} ($\mathrm{strict}<\mathrm{Scott}<\mathrm{TB}<\mathrm{aperiodic}$) while $\leq_k$ on \texttt{Articulating.Corner} is a \textbf{diamond} (t, f incomparable), so the conflation is \emph{not} an order-isomorphism. It must be carried by the Klein-four \textbf{gauge} ($\sim$, $-$), which the regime side did not carry as operations until \cref{thm:regime-spin} transported one generator (\texttt{negSwap}) across and found it acts as a 4-cycle skipping the $\kappa$-chain order, not as the chain order itself --- a real but partial answer, one generator of a group, not the group. Red until the remaining \texttt{D8} generators are transported and checked against the regime side's own operations (if any exist to check against), and \texttt{quartet\_recurrence} is extended to six readings.
\end{theorem}
\begin{theorem}[The FCA derivation closure is a settling-shape closure operator]\label{thm:fca-closure-laws}\lean{Obsmetrologia.FCALambek.fca_closure_laws}\leanok\uses{thm:closure-operator}
\texttt{fcaClosureOp} / \texttt{fca\_closure\_laws}: for any formal context $I$, the object-side derivation closure $A\mapsto A''$ is \emph{extensive}, \emph{monotone}, and \emph{idempotent} --- exactly the three laws of \texttt{Charge.Closure} (\texttt{infl}/\texttt{mono}/\texttt{idem}) that the settling morphism iterates. Proved by reuse of FCA's own \texttt{subA\_closeA} / \texttt{closeA\_mono} / \texttt{closeA\_idem}; the open node below upgrades this shape-match to an operator \emph{identity}.
\end{theorem}
\begin{theorem}[The FCA Galois closure is the settling closure --- as an identity]\label{thm:fca-closure-eq-settling}\notready\uses{thm:fca-closure-laws}
\emph{(Open frontier --- stated, not formalized.)} \texttt{fcaClosureOp} / \texttt{fca\_closure\_laws} prove the FCA derivation closure $A \mapsto A''$ has the settling closure's \emph{shape} --- extensive, monotone, idempotent, the three laws of \texttt{Charge.Closure}. But ``computing concepts \emph{is} running the settling closure'' is currently a shape-match, not an \textbf{operator identity}: that $A \mapsto A''$ literally equals the settling $\Box\Diamond$ on a shared carrier is unproven. Red until the two carriers are unified and the operators shown equal pointwise (whereupon FCA's fundamental theorem becomes a \emph{corollary} of settling-closure completeness, not a parallel development).
\end{theorem}
\begin{theorem}[Abrams' correspondence on the worked Frobenius algebra]\label{thm:frobenius-tqft-instance}\lean{Obsmetrologia.Cohomology.FrobeniusTQFT.commutative_frobenius_is_tqft}\leanok\uses{thm:settling-cohomology-h1-vanishes, thm:k4-group}

This is the \emph{instance}; the general categorical equivalence is the open node below.
\end{theorem}
\begin{theorem}[The general Abrams equivalence]\label{thm:abrams-equivalence-general}\notready\uses{thm:frobenius-tqft-instance}
\emph{(Open frontier --- stated, not formalized.)} \texttt{frobeniusTqftCorrespondence} / \texttt{commutative\_frobenius\_is\_tqft} prove Abrams' correspondence on the \emph{single worked algebra} $Z = \mathbb{Z}[x]/(x^2)$: its induced 2D TQFT has circle value $Z$ and matching generators, with the round-trip recovering $Z$ (all by \texttt{rfl}). Red until the equivalence of categories is constructed, not just instantiated.
\end{theorem}
