\section{Reading the draft back is M\"obius inversion}

\begin{theorem}[Recognizability is M\"obius inversion]\label{thm:recognizability-is-mobius}\notready\uses{thm:quasiperiodic-weave-recognizable, thm:causal-manifoldlikeness}
\emph{(Open frontier --- stated, not formalized.)} Reading a finished fabric back to its
generating warp programme (recognizability, \cref{thm:quasiperiodic-weave-recognizable}) is
M\"obius inversion of the interlacement incidence order: if the cloth is $G\star$ draft
(propagator), the draft is $\mu\star$ cloth, with $\mu$ the causal M\"obius function
(\texttt{Obsmetrologia.CausalSets.Incidence.Mobius}, \texttt{mobius\_eq\_hallSum}: $\mu$ is the
reduced Euler characteristic of the order complex). Recognizability is the existence and
uniqueness of that inverse.
\end{theorem}

\begin{theorem}[A M\"obius-band textile is mirror-holonomy]\label{thm:mobius-band-holonomy}\notready\uses{thm:cloth-faces-negation-mirror}
\emph{(Open frontier --- stated, not formalized.)} A M\"obius-band textile (the
bead-crochet M\"obius rope) is the over/under mirror $\sim$ carried around a loop with
nontrivial holonomy: the negation involution composed once around the band is not the
identity but the face-swap, so the band has one face, not two. Non-orientable
interlacement \emph{is} mirror-holonomy --- a global sign defect, the $\top$-seam that
cannot be combed away.
\end{theorem}
