\section{Curvature defects are the Einstein--Hilbert action}

\begin{theorem}[A curvature defect is the Benincasa--Dowker action density]\label{thm:curvature-defect-bd-action}\notready\uses{thm:fabric-is-propagator, thm:weave-entropy-collapse}
\emph{(Open frontier --- stated, not formalized.)} The combinatorial curvature
$\kappa(v) = 2\pi - \sum_{f\ni v}\theta_{v,f}$ concentrated at a textile/beadwork defect
(a pentagon carries $+\kappa$, a heptagon $-\kappa$ in a hexagonal net) is the
Benincasa--Dowker discrete Einstein--Hilbert action density (\texttt{Spacetime.Action.EinsteinHilbertAction.bdAction},
the $\mu$-weighted alternating chain sum inside \cref{thm:chromatic-lorentzian-action}). A
flat plain weave has zero action (\cref{thm:weave-entropy-collapse}: maximal coherence,
zero curvature); the action lives entirely at the defects.
\end{theorem}

\begin{theorem}[Textile Gauss--Bonnet]\label{thm:textile-gauss-bonnet}\notready\uses{thm:curvature-defect-bd-action}
\emph{(Open frontier --- stated, not formalized.)} Total defect curvature equals
$2\pi\chi$ with $\chi = V - E + F = 2 - 2g$ (discrete Gauss--Bonnet): the signed count of
pentagon-minus-heptagon defects fixes the fabric's genus. A beaded sphere requires exactly
twelve pentagonal defects; a torus exactly zero. The Einstein--Hilbert action of a
charge-coloured tiling is a topological invariant of the cloth.
\end{theorem}
