  (let* ((nodes
          (org-roam-db-query
           [:select [*]
            :from nodes
            :where (and
  		  (like properties '"%form%text catalog%")
  		  (like properties '"%public%true%"))]))
         (nodes-with-date
  	(cl-loop for node in nodes
  		 for file = (nth 1 node)
  		 for date = (with-temp-buffer
  			      (when (and file (file-readable-p file))
  				(insert-file-contents file)
  				(org-mode)
  				(goto-char (point-min))
  				(let ((date-entry (cdr (assoc "DATE" (org-collect-keywords '("DATE"))))))
  				  (and date-entry (string-trim (car date-entry))))))  				
  		 collect (append node (list date))))
         (sorted-nodes
  	(sort nodes-with-date
  	      (lambda (a b)
  		(let ((ad (ignore-errors (org-time-string-to-time (nth 11 a))))
  		      (bd (ignore-errors (org-time-string-to-time (nth 11 a)))))
  		  (and ad bd (time-less-p bd ad)))))))
    (mapconcat
     (lambda (node)
       (format "- [[id:%s][%s]]" (car node) (nth 8 node)))
     sorted-nodes
     "\n"))
