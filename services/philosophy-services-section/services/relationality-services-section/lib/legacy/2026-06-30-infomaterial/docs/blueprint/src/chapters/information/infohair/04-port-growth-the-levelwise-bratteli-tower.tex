\section{Port-growth: the levelwise Bratteli tower}

A textile grows levelwise --- knit rows, bead courses --- and that growth is a Bratteli diagram with
incidence \texttt{substMatrix}. The tower and its composition law are green in
\texttt{Infomaterial.Persistence.Aperiodic.PortGrowth}.  The infinite-depth
self-similarity is a separate StepN colimit claim.

\begin{theorem}[The Step-$n$ shift chain has a colimit]\label{thm:stepn-shift-chain-colimit}\lean{Infomaterial.Persistence.Aperiodic.StepN.step_shift_chain_colimit}\leanok\uses{thm:stepn-co-increment, thm:universality-se-class}
The step-shift sends each finite step-$n$ moduli point into the next depth,
the embeddings are compatible, and the directed colimit identifies a point
across all step depths.
\end{theorem}

\begin{theorem}[The Step-$n$ colimit is an RG fixed point]\label{thm:stepn-colimit-rg-fixed-point}\lean{Infomaterial.Persistence.Aperiodic.StepN.step_n_colimit_rg_fixed_point}\leanok\uses{thm:stepn-shift-chain-colimit}
At infinite step depth, the step-shift flow is the identity: advancing one
more depth is invisible in the colimit.  This is the formal fixed-point
reading of the self-similar StepN tower.
\end{theorem}

\begin{proposition}[The port-growth tower]\label{thm:port-growth}\lean{Infomaterial.Persistence.Aperiodic.portGrowth}\leanok\uses{thm:metallic-family-cayley-hamilton}
$\texttt{portGrowth}\,M\,k = M^k$: the level-$k$ reachability of a per-level port-incidence matrix
$M$, whose entry counts the thread-routes a $k$-level growth opens. A stationary (constant-$M$)
Bratteli diagram --- the self-similar textile, with $M = \texttt{substMatrix}$.
\end{proposition}

\begin{theorem}[The Bratteli composition law]\label{thm:port-growth-telescopes}\lean{Infomaterial.Persistence.Aperiodic.portGrowth_telescopes}\leanok\uses{thm:port-growth}
$\texttt{portGrowth}\,M\,(a+b) = \texttt{portGrowth}\,M\,a \cdot \texttt{portGrowth}\,M\,b$: growing
$a+b$ levels factors as growing $a$ then $b$. The cocycle that makes the tower a levelwise system
(matrix-power additivity, proved here).
\end{theorem}

\begin{theorem}[Plastic-number renormalization]\label{thm:port-growth-renorm}\lean{Infomaterial.Persistence.Aperiodic.portGrowth_renorm_padovan}\leanok\uses{thm:port-growth, thm:stepn-chu-cayley-hamilton, thm:stepn-colimit-rg-fixed-point}
For the Padovan weave, $M^3 = M + I$: growing three levels equals growing one and holding the
identity. The inflation fixed point (self-similarity) of the smallest-Pisot textile, read on its
port-growth tower.
\end{theorem}

\begin{theorem}[The Bratteli edges are FOUR-coloured]\label{thm:port-growth-chu}\lean{Infomaterial.Persistence.Aperiodic.portGrowth_edges_chu}\leanok\uses{thm:port-growth, thm:four-object}
Each port-incidence entry of the level matrix is the Chu space's incidence bit, so the levelwise
growth's edges carry the bilattice colour: a port-growth tower is a bilattice-coloured Bratteli
diagram.
\end{theorem}
