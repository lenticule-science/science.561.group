\section{Proper time and continuum convergence}

\begin{theorem}[Proper time is the order-intrinsic longest chain]\label{thm:proper-time-longest-chain}\lean{Infomaterial.Relational.Spacetime.Causal.Reachability.diamond_properTime}\leanok\uses{thm:causal-settling-transitive-closure}
The genuinely \emph{geometric}, manifold-free invariant of a causal set: the timelike distance
$\tau(x,y)$ is the \textbf{longest $\prec$-chain} from $x$ to $y$ (the discrete proper time), $0$ when
$x\not\prec y$. Saturated at fuel $|{\rm events}|$ (\texttt{properTime}). On the diamond
$\tau(0,3)=2$ (two-link geodesic), the covering links $=1$, and the spacelike middle pair has
$\tau(1,2)=0$ (\texttt{diamond\_properTime}, \texttt{diamond\_spacelike\_zero}). With the interval
\textbf{volume} $|I(x,y)|$ (\texttt{Spacetime.Causal.Interval.intervalCard}) this is Sorkin's \emph{order $+$
number $=$ geometry}: causal order fixes the conformal structure, the count fixes volume --- no
embedding manifold. (This is intrinsic geometry, distinct from the curvature-magnitude frontier
\cref{thm:sym-trace-is-ricci-scalar}.)
\end{theorem}

\begin{theorem}[Reverse (Lorentzian) triangle inequality on the diamond]\label{thm:reverse-triangle-diamond}\lean{Infomaterial.Relational.Spacetime.Causal.Reachability.diamond_reverse_triangle}\leanok\uses{thm:proper-time-longest-chain}
The signature of Lorentzian geometry, realized concretely: $\tau(0,1)+\tau(1,3)\le\tau(0,3)$ on the
diamond ($1+1\le 2$, with \emph{equality} because $0\prec1\prec3$ is a geodesic). Proper time is
\emph{maximized} along geodesics --- the timelike opposite of the metric triangle inequality. Proved
by \texttt{decide} at the diamond; the general statement is \cref{thm:reverse-triangle-general}.
\end{theorem}

\begin{theorem}[Proper time grows by a tick per causal step]\label{thm:proper-time-step}\lean{Infomaterial.Relational.Spacetime.Causal.Reachability.longestChain_step}\leanok\uses{thm:proper-time-longest-chain}
The single-step heart of the reverse triangle, now green: if $x\prec y$ and $z$ is a successor of $x$
with $x\prec z\preceq y$ (an event), then $1 + \mathrm{longestChain}\,n\,z\,y \le
\mathrm{longestChain}\,(n{+}1)\,x\,y$ --- one link toward the future adds at least one tick of proper
time (\texttt{longestChain\_step}), since the foldr-max over successors dominates the member $z$
(\texttt{foldr\_max\_ge\_mem}). The per-step monotonicity the general inequality iterates.
\end{theorem}

\begin{theorem}[Reverse triangle inequality, general]\label{thm:reverse-triangle-general}\lean{Infomaterial.Relational.Spacetime.Causal.Reachability.reverse_triangle_general}\leanok\uses{thm:proper-time-step}
For every finite causal order (transitive, irreflexive, predecessors listed) and all
$x\preceq z\preceq y$, $\tau(x,z)+\tau(z,y)\le\tau(x,y)$ --- proper time is maximized along geodesics,
the timelike opposite of the metric triangle inequality. Proved by iterating the single-step bound
(\cref{thm:proper-time-step}) along the first segment (\texttt{longestChain\_concat}, induction on
fuel, prepending \texttt{properTime\_lt\_step} at each successor), transferred to the saturated
$\mathrm{properTime}$ by \emph{fuel stabilization}: $\mathrm{longestChain}$ is constant once the fuel
reaches the apex rank (\texttt{longestChain\_fuel\_stable}), and fuel $|{\rm events}|$ already
saturates because the past count is a topological index bounded by $|{\rm events}|$
(\texttt{longestChain\_saturated}, reusing \texttt{Spacetime.Incidence.Mobius.pastCount\_topo}). The diamond
instantiates it non-vacuously (\texttt{diamond\_reverse\_triangle\_general}).
\end{theorem}

\begin{theorem}[Brightwell--Gregory: proper time converges to geodesic distance]\label{thm:brightwell-gregory-continuum}\notready\uses{thm:reverse-triangle-general, thm:myrheim-meyer-dimension}
\emph{(Open frontier --- the continuum bridge, stated, not formalized.)} For a faithful Poisson
sprinkling at density $\rho$ into a $d$-dimensional Lorentzian manifold, the rescaled longest chain
$\tau(x,y)/(m_d\,\rho^{1/d})$ converges (a.s.) to the continuum geodesic proper time between the
sprinkled events (Brightwell--Gregory 1991). This is the theorem that makes the intrinsic
\cref{thm:proper-time-longest-chain} \emph{the} geometry rather than mere combinatorics. Stating it
needs the reals, the sprinkling measure, and a model spacetime --- none present. Red until a sprinkling
measure over internal reals and a continuum proper-time target exist.
\end{theorem}
