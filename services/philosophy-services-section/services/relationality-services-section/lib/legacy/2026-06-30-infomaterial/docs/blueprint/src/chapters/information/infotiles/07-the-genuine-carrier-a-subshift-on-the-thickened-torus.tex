\section{The genuine carrier: a $\mathbb{Z}^2$ subshift on the thickened torus}

\Cref{thm:weave-2d-sofic} requires a carrier not supplied by the present
$1$D \texttt{Path} development.  This subgraph isolates that missing fiber: the
$\mathbb{Z}^2$ subshift, its strand-transitive homogeneity, and its
thickened-torus realization.

\begin{theorem}[The thickened-torus $\mathbb{Z}^2$ carrier]\label{thm:thickened-torus-z2-carrier}\notready\uses{thm:weave-2d-sofic, thm:aperiodic-settle-frame}
\emph{(Open frontier --- stated, not formalized.)} The carrier behind \cref{thm:weave-2d-sofic}
is the bi-infinite configuration space $A^{\mathbb{Z}^2}$ with a finite allowed-neighbourhood
set $\mathcal{F}\subseteq A^N$, quotiented by a rank-two lattice $\Lambda\cong\mathbb{Z}^2$ so
that $x/\Lambda$ lives in the thickened torus $T^2\times I$. The claim: this $2$D subshift is
a genuine settling frame in the $\kappa\geq\omega_1$ aperiodic regime
(\cref{thm:aperiodic-settle-frame}), its admissible configurations the settled fixed points of
the neighbourhood constraint. Requires a $\mathbb{Z}^2$-indexed
\texttt{Infomaterial.Persistence.Aperiodic.TorusSubshift} carrier the present $\mathbb{Z}$-indexed
\texttt{Path} does not provide.
\end{theorem}

\begin{theorem}[Isonemality is strand-transitive homogeneity]\label{thm:isonemal-layergroup-transitivity}\notready\uses{thm:thickened-torus-z2-carrier, thm:klein-four-symmetry}
\emph{(Open frontier --- stated, not formalized.)} Roth's isonemality --- the fabric
symmetry group (a crystallographic layer group) acts transitively on strands --- is the
homogeneity condition on the $\mathbb{Z}^2$ carrier under which a shift-invariant measure,
hence a measured entropy, is well-defined. The $42$ isonemal types are the orbit classes of
the layer-group action on the charge-coloured torus subshift, and the over/under involution
$\sim$ (\cref{thm:klein-four-symmetry}) is the colour-reversing element distinguishing
colour-preserving from colour-reversing perfect colourings. Requires the layer-group action
on the carrier of \cref{thm:thickened-torus-z2-carrier}.
\end{theorem}

\begin{theorem}[A doubly-periodic draft is a charge-coloured link in $T^2\times I$]\label{thm:textile-link-on-torus}\notready\uses{thm:thickened-torus-z2-carrier, thm:miscrossing-top}
\emph{(Open frontier --- stated, not formalized.)} The Grishanov--Meshkov--Omelchenko
reading --- a doubly-periodic fabric is a link diagram on a torus --- is the geometric
realization of the $\mathbb{Z}^2$ carrier: each crossing of the link is a charge cell
(\cref{thm:miscrossing-top}), and the link's two-strand over/under data \emph{is} the
charge-colouring of \cref{thm:thickened-torus-z2-carrier}. The diagram lives in $T^2\times I$
precisely because the configuration is $\Lambda$-periodic. Requires the embedding of the
torus subshift into a thickened-torus link complex.
\end{theorem}
