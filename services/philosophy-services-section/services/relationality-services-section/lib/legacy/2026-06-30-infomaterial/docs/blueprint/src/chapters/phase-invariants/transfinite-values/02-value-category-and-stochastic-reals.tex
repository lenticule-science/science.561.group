\subsection{Value category and stochastic internal reals}

\begin{proposition}[The value category]\label{thm:value-category}\lean{Infomaterial.Positing.Value.ValueCategory}\leanok\uses{thm:four-bilattice, thm:strict-finite}
A \texttt{ValueCategory}: a carrier with a reflexive, transitive, antisymmetric order and a bottom element --- the value layer that Markov-kernel stochastic settling runs over.
\end{proposition}

\begin{theorem}[The real stochastic interval is a value category]\label{thm:real-stochastic-interval-value-category}\lean{Infomaterial.Probability.RealStochastic.realUnitInterval_is_value_category}\leanok\uses{thm:value-category, thm:reals-close-value-settling-hub}
\texttt{Probability/Stochastic.lean} was abstract because it predated internal reals: only a
\texttt{StochSemiring} and a \texttt{Nat} witness, no genuine $[0,1]$. With the internally derived
constructive reals (\texttt{Real.Internal}) present, the unit interval is built: \texttt{UnitReal},
the subtype $\{x:\mathbb{R} \mid 0\le x\le 1\}$, is a genuine \texttt{ValueCategory} --- reflexive,
transitive, and antisymmetric under the real order (antisymmetry from the reals' antisymmetry plus
proof irrelevance on the bounds), with seed $0$. This is the value layer for stochastic settling
over real probabilities, replacing the abstract semiring. A further refinement (red) instantiates the
actual Markov-kernel stochastic settling over it and proves the probability-normalization laws.
\end{theorem}

\begin{theorem}[Finite thermodynamic transition production]\label{thm:finite-thermodynamic-transition-production}\lean{Infomaterial.Thermodynamics.Temperature.LiftedThermoTransition.produces_creation_landauer_and_discharge}\leanok\uses{thm:real-stochastic-interval-value-category, thm:stochastic-is-value-settling, thm:measure-is-valuation}
A lifted thermodynamic transition retaining its witnesses carries positive
held-\(\top\) creation, balanced stochastic flux, positive Landauer cost, and
positive hard-flow discharge.  This is the finite churn package; it does not yet
construct a real-valued free-energy functional.
\end{theorem}

stationary settling equilibria. Missing bridge: a measure/expectation layer over internal reals
plus a stochastic closure operator compatible with the settling order. (Target:
\texttt{Infomaterial.Flow.positiveTemperatureFreeEnergy}.)
\end{theorem}
