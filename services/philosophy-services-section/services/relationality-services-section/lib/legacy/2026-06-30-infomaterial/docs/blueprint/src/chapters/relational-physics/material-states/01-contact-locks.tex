\subsection{Contact locks}

\begin{proposition}[Keratin lock]\label{thm:keratin-lock}\lean{Infomaterial.Relational.Material.Lock}\leanok\uses{thm:tarski-sheaf-closure}
A keratin contact lock is an inflationary monotone closure step on a complete
lattice of contact states.
\end{proposition}

\begin{theorem}[Keratin settling is the least compatible lock]\label{thm:keratin-settle-lock}\lean{Infomaterial.Relational.Material.settle_lock_is_least_compatible}\leanok\uses{thm:keratin-lock}
Running lock histories over a keratin complex yields the least compatible
settled contact assignment above the seed.
\end{theorem}
