\subsubsection{Causal net region theory}

\begin{theorem}[Causal net region theory]\label{thm:causal-net}\lean{Infomaterial.Relational.Spacetime.Causal.Net.causal_net_region_theory}\leanok\uses{thm:cone-field-geometry}
Causal regions form an inclusion preorder; closed diamonds are monotone in
their endpoints; spacelike-separated regions are symmetric and disjoint. An
independent causal net transports spacelike separation into independence of
local data in both directions.
\end{theorem}
