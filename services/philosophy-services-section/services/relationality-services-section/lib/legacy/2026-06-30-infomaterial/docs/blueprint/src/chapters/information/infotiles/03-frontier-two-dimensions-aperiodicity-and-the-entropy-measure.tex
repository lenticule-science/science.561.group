\section{Frontier: two dimensions, aperiodicity, and the entropy measure}

\begin{theorem}[Two-dimensional isonemal soficity]\label{thm:weave-2d-sofic}\notready\uses{thm:weave-entropy-collapse, thm:charge-coloured-sofic, thm:aperiodic-mld-classification}
\emph{(Open frontier --- stated, not formalized.)} Claim-site: the finite-swatch
\texttt{List} model lifts to a bi-infinite draft $x\in A^{\mathbb{Z}^2}$ with finite
neighbourhood constraints.  The lifted draft is a $2$D sofic tiling, its charge
factor is \cref{thm:charge-coloured-sofic}, and strand transitivity supplies the
homogeneity needed for measured entropy.  The aperiodic classification incidence is
covered by \cref{thm:aperiodic-mld-classification}; the residue is the
$\mathbb{Z}^2$ thickened-torus carrier.
\end{theorem}

\begin{theorem}[Recognizability of the quasiperiodic weave]\label{thm:quasiperiodic-weave-recognizable}\notready\uses{thm:weave-draft-sofic, thm:stepn-growth-degree, thm:weave-is-metallic}
\emph{(Open frontier --- stated, not formalized.)} A Sturmian or metallic-mean weave
row is the forgetful image of a decorated finite-type cover.  Recognizability says
that the finished swatch determines its generating warp programme uniquely.  The
existing \texttt{SoficShift} carrier proves the golden case
(\texttt{fib\_deep\_sofic}); the residue is the primitive-substitution
recognizability map.
\end{theorem}

\begin{theorem}[Tiles measured entropy of a draft]\label{thm:weave-infotiles-entropy}\notready\uses{thm:weave-entropy-collapse, thm:charge-coloured-sofic, thm:weave-2d-sofic}
\emph{(Open frontier --- stated, not formalized.)} The measured entropy of a
charge-coloured draft is the topological entropy of its underlying sofic subshift.
\Cref{thm:weave-entropy-collapse} is the zero-rate finite witness; the open fiber is
an entropy/diffraction measure on the charge-coloured $2$D tiling.
\end{theorem}
