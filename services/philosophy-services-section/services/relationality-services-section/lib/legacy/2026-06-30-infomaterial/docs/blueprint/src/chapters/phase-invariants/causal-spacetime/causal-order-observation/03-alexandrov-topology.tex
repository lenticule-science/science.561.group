\subsection{Alexandrov topology}

\begin{theorem}[Causal Alexandrov topology recovers the cones]\label{thm:causal-alexandrov-topology}\lean{Infomaterial.Relational.Spacetime.Causal.Alexandrov.causal_alexandrov_recovers_cones}\leanok\uses{thm:cone-field-geometry}
The Alexandrov topology generated by the finite causal order recovers the
causal structure: specialization is reflexive causal precedence, strict
reachability is specialization plus inequality, and principal future cones
are open.
\end{theorem}
