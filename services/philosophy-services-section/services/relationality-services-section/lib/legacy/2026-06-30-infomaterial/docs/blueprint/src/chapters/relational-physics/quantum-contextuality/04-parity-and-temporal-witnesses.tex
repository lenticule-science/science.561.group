\subsection{Parity and temporal witnesses}

\begin{theorem}[The Mermin--Peres magic square is strongly contextual]\label{thm:magic-square-strong-contextual}\lean{Infomaterial.Relational.Quantum.magicSquare_strongly_contextual}\leanok\uses{thm:possibilistic-model}
The magic square gives a state-independent all-versus-nothing obstruction:
all six line contexts are locally satisfiable, but no global assignment
satisfies their parity constraints.
\end{theorem}

\begin{theorem}[Leggett--Garg violates macrorealism]\label{thm:leggett-garg-macrorealism-violated}\lean{Infomaterial.Relational.Quantum.lgPerfect_macrorealism_violated}\leanok\uses{thm:possibilistic-model}
The temporal Leggett--Garg support model is strongly contextual: the pairwise
time constraints cannot come from one definite trajectory.
\end{theorem}

\begin{theorem}[The Leggett--Garg quantum value is not macrorealist]\label{thm:leggett-garg-real-separation}\lean{Infomaterial.Relational.Quantum.lg_quantum_value_not_macrorealist}\leanok\uses{thm:leggett-garg-macrorealism-violated}
Over the internal real/rational value layer, no definite macrorealist
trajectory realizes the \(3/2\) quantum value.
\end{theorem}
