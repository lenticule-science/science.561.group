\subsection{Finite kinetic gas}

\begin{theorem}[Finite Boltzmann collision closure]\label{thm:finite-boltzmann-collision-lfp}\lean{Infomaterial.Relational.Kinetic.finite_boltzmann_collision_lfp}\leanok\uses{thm:closure-rule-lfp}
The finite collision rule system has a least fixed-point closure, giving the
Boltzmann-collision projection of rule-system settling.
\end{theorem}

\begin{theorem}[Discrete collisions preserve invariants]\label{thm:discrete-collision-conservation}\lean{Infomaterial.Relational.Kinetic.evolution_conserves}\leanok\uses{thm:finite-boltzmann-collision-lfp}
Evolution by physical discrete collisions preserves every collision invariant,
including the number, momentum, and energy witnesses formalized in the module.
\end{theorem}

\begin{theorem}[Molecular chaos glues factorized joints]\label{thm:molecular-chaos-factorization}\lean{Infomaterial.Relational.Kinetic.glue_factorizes}\leanok\uses{thm:finite-boltzmann-collision-lfp}
The product gluing of two marginals is factorized. The companion
nonfactorized witness keeps the propagation-of-chaos frontier separate from
the finite gluing fact.
\end{theorem}
