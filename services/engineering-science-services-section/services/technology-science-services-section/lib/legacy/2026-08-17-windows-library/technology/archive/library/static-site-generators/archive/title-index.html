{{- /* Build a map of lowercased titles/aliases/defines/slugs → RelPermalink. Called via partialCached so this runs exactly once per build. Also indexes by URL-slug so frontmatter values like "philosophy-of-endeavor" resolve to the matching page. */ -}} {{- $index := dict -}} {{- range site.Pages -}} {{- $url := .RelPermalink -}} {{- /* slug = last URL segment (handles both regular pages and _index.md sections) */ -}} {{- $slug := path.Base (strings.TrimSuffix "/" $url) -}} {{- with $slug -}} {{- $lower := lower . -}} {{- if and (ne $lower "") (ne $lower "/") (not (index $index $lower)) -}} {{- $index = merge $index (dict $lower $url) -}} {{- end -}} {{- end -}} {{- with .Title -}} {{- $lower := lower . -}} {{- if not (index $index $lower) -}} {{- $index = merge $index (dict $lower $url) -}} {{- end -}} {{- end -}} {{- with .Params.aliases -}} {{- range . -}} {{- $lower := lower . -}} {{- if not (index $index $lower) -}} {{- $index = merge $index (dict $lower $url) -}} {{- end -}} {{- end -}} {{- end -}} {{- with .Params.defines -}} {{- if reflect.IsSlice . -}} {{- range . -}} {{- $lower := lower . -}} {{- if not (index $index $lower) -}} {{- $index = merge $index (dict $lower $url) -}} {{- end -}} {{- end -}} {{- else -}} {{- $lower := lower . -}} {{- if not (index $index $lower) -}} {{- $index = merge $index (dict $lower $url) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $index -}}
